foreach (it ${LOGGER_H_SRC})
    get_filename_component(_outfile ${it} NAME)
    get_filename_component(_infile ${it} ABSOLUTE)
    configure_file(${_infile} ${_outfile} COPYONLY)
    set(LOGGER_H_SOURCES, ${LOGGER_H_SOURCES} ${_outfile})
endforeach(it)

qore_wrap_qpp_value(LOGGER_QPP_SOURCES DOXLIST _dox_src ${LOGGER_QPP_SRC})
add_custom_target(LOGGER_QPP_GENERATED_FILES DEPENDS qpp ${LOGGER_QPP_SOURCES} ${LOGGER_H_SOURCES})

# prepare logger module target
set(logger_module_name "logger_bin")

if (SINGLE_COMPILATION_UNIT)
    # Generate SCU file in build directory with absolute paths
    set(LOGGER_SCU_PATH ${CMAKE_CURRENT_BINARY_DIR}/logger_scu.cpp)

    set(_scu_content "// Auto-generated logger single-compilation-unit file\n")
    set(_scu_content "${_scu_content}// Generated by cmake — do not edit manually\n\n")

    # Direct C++ sources
    foreach(_f ${LOGGER_CPP_SRC})
        get_filename_component(_abs ${_f} ABSOLUTE)
        set(_scu_content "${_scu_content}#include \"${_abs}\"\n")
    endforeach()
    set(_scu_content "${_scu_content}\n")

    # QPP-generated sources (already absolute paths in build dir)
    foreach(_f ${LOGGER_QPP_SOURCES})
        set(_scu_content "${_scu_content}#include \"${_f}\"\n")
    endforeach()

    file(WRITE ${LOGGER_SCU_PATH} "${_scu_content}")

    add_library(${logger_module_name} MODULE ${LOGGER_SCU_PATH})

    # Declare file-level dependencies so cmake rebuilds when sources change
    set(_scu_deps ${LOGGER_QPP_SOURCES})
    foreach(_f ${LOGGER_CPP_SRC})
        get_filename_component(_abs ${_f} ABSOLUTE)
        list(APPEND _scu_deps ${_abs})
    endforeach()
    set_source_files_properties(${LOGGER_SCU_PATH} PROPERTIES
        OBJECT_DEPENDS "${_scu_deps}"
    )
else (SINGLE_COMPILATION_UNIT)
    add_library(${logger_module_name} MODULE ${LOGGER_CPP_SRC} ${LOGGER_QPP_SOURCES})
endif (SINGLE_COMPILATION_UNIT)

add_dependencies(${logger_module_name} libqore LOGGER_QPP_GENERATED_FILES)

qore_binary_module_qore(${logger_module_name} "${VERSION}" "/${VERSION}")
